# CS 101–103 300-Script Stress Corpus & 2-Axis Gap Runner Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a 300-script university CS 101–103 Memelang stress corpus under `tests/corpus/` in the project author's voice, paired with deterministic fixtures, `manifest.json`, and a 2-axis (`examples/` vs `src/` language) gap-attribution runner (`scripts/run_corpus.py`).

**Architecture:** All 300 `.meme` test scripts, shared data fixtures, and `manifest.json` live strictly under `tests/corpus/` so `examples/` stays untouched as Memelang's shipped syntax/library extensibility directory. `scripts/run_corpus.py` imports `memelang` in-process (loading GGUF weights once per run), evaluates filtered slices against an isolated temporary lockfile, and buckets results across 8 outcome states and Memelang's two extensibility axes (`examples` syntax/idiom gaps vs `language` IR/codegen gaps). Fast `<1s` `pytest` checks in `tests/test_corpus_manifest.py` enforce all 300-script quotas, uniqueness, fixture resolution, and runner logic.

**Tech Stack:** Python 3.13+, Pydantic v2, `pytest`, `memelang` (`src/memelang/`).

**Spec:** `docs/superpowers/specs/2026-11-03-cs101-103-corpus-design.md`

## Global Constraints

- Zero extra third-party dependencies beyond existing `pyproject.toml` (`pydantic>=2.9`, `httpx>=0.27`).
- `examples/` MUST NOT be modified or receive test corpus files; all corpus assets live under `tests/corpus/` and `scripts/run_corpus.py`.
- Exact corpus counts: `300` scripts total (`100` in `cs101` IDs `1..100`, `100` in `cs102` IDs `101..200`, `100` in `cs103` IDs `201..300`).
- Exact per-course style split: `80` `"clean"`, `10` `"typo"`, `10` `"ambiguous"` per course (`240 / 30 / 30` global).
- Every script must be a genuinely distinct CS problem written in the author's Memelang voice (`this runs in the shell`, flat CS 101 scripts or `class <kebab-name>` + `define function ... with ...` + `define function main`, 1-based `item 1`/`item last`, declarative + step-by-step procedural English).
- `uv run pytest` must remain fast (`<25s` total) and never invoke the 3B GGUF LLM over the 300-script corpus.

## Review Focus

- Running `scripts/run_corpus.py` from the repository root must use a temporary lockfile and never create, read, or mutate `./build.lock`.
- Relative fixture paths entered via `stdin` (such as `../fixtures/numbers.json` from `tests/corpus/cs102/*.meme`) must resolve against the script's parent directory (`source_dir`), matching `meme run`.
- Prompt text echoed to stdout (e.g., `"Enter item 1? "`) must not cause a false-positive `PASS` when checking `expected_stdout_contains` (e.g., expecting `"1"` when actual program output after prompts is `"0"`).
- Invalid `--ids` filters (such as `"10-5"`, `"0"`, or `"abc"`) must produce a clean CLI error (exit code `2`) rather than an unhandled traceback or silent 0-script pass.
- No two scripts among the 300 may share identical normalized content or collapse to numeric-only constant clones when digits and whitespace are normalized.

```python
import importlib.util
from pathlib import Path
import pytest

from memelang.ir.schema import (
    AmbiguityCandidate,
    AmbiguityDiagnostic,
    FuncDefStmt,
    FuncParam,
    LiteralExpr,
    PrintStmt,
    ReadInputExpr,
    ReadJsonListExpr,
    ReturnStmt,
    SortCollectionStmt,
    SynthesisResult,
    VarDeclStmt,
    VarRefExpr,
)
from memelang.llm.base import LLMProviderError
from memelang.lockfile.manager import LockfileManager

REPO_ROOT = Path(__file__).resolve().parents[1]


def _load_runner_module():
    runner_path = REPO_ROOT / "scripts" / "run_corpus.py"
    spec = importlib.util.spec_from_file_location("run_corpus", runner_path)
    assert spec is not None and spec.loader is not None
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def test_shared_fixtures_exist_and_valid():
    fix_dir = REPO_ROOT / "tests" / "corpus" / "fixtures"
    expected = {
        "numbers.json",

def test_parse_id_filter_valid_and_invalid():

def test_evaluate_script_buckets_prompt_stripping_source_dir_and_lockfile_isolation(tmp_path):
    rc = _load_runner_module()
    sub = tmp_path / "tests" / "corpus" / "cs102"
    fix = tmp_path / "tests" / "corpus" / "fixtures"
    sub.mkdir(parents=True)
    fix.mkdir(parents=True)
    (fix / "numbers.json").write_text("[10, 50, 3]\n", encoding="utf-8")

    s_path = sub / "sample.meme"
    s_path.write_text('print "x"\n', encoding="utf-8")

    def _make_spec(expected_status="ok", expected_out="1", stdin="0\n", ex_gaps=None, lang_gaps=None):
        return rc.CorpusScriptSpec(
            id=1,
            slug="001_sample",
            path=str(s_path.relative_to(tmp_path).as_posix()),
            course="cs102",
            topic="test",
            title="Sample",
            style="clean" if expected_status == "ok" else "ambiguous",
            stdin=stdin,
            expected_status=expected_status,
            expected_stdout_contains=expected_out,
            feature_tags=["print"],
            extensibility_gaps=rc.ExtensibilityGaps(
                examples=ex_gaps or [], language=lang_gaps or []
            ),
        )

    class StubProvider:
        name = "stub"
        def __init__(self, fn):
            self._fn = fn
        def synthesize(self, text: str) -> SynthesisResult:
            return self._fn(text)

    lock_mgr = LockfileManager(tmp_path / "isolated.lock")

    # 1. Prompt echo ("Item 1? ") must NOT falsely satisfy expected_stdout_contains="1" when answer is 0
    prompt_prog = SynthesisResult(
        items=[
            VarDeclStmt(
                name="v",
                var_type="Int",
                value=ReadInputExpr(prompt="Item 1? ", target_type="Int"),
            ),
            PrintStmt(expressions=[VarRefExpr(name="v")]),
        ]
    )
    res_mismatch = rc.evaluate_script(
        _make_spec(expected_out="1", stdin="0\n", ex_gaps=["idiom_a"]),
        StubProvider(lambda _: prompt_prog),
        "python",
        lock_mgr,
        repo_root=tmp_path,
    )
    assert res_mismatch.bucket == "OUTPUT_MISMATCH"

    # 2. Relative fixture `../fixtures/numbers.json` resolves via script parent `source_dir`
    json_prog = SynthesisResult(
        items=[
            VarDeclStmt(
                name="fp",
                var_type="String",
                value=ReadInputExpr(prompt="Path? ", target_type="String"),
            ),
            VarDeclStmt(
                name="nums",
                var_type="List<Float>",
                value=ReadJsonListExpr(filepath=VarRefExpr(name="fp")),
            ),
            SortCollectionStmt(target="nums", direction="ascending"),
            PrintStmt(expressions=[VarRefExpr(name="nums")]),
        ]
    )
    res_pass = rc.evaluate_script(
        _make_spec(expected_out="[3, 10, 50]", stdin="../fixtures/numbers.json\n"),
        StubProvider(lambda _: json_prog),
        "python",
        LockfileManager(tmp_path / "isolated2.lock"),
        repo_root=tmp_path,
    )
    assert res_pass.bucket == "PASS"

    # 3. Synthesis error, static verify gap, expected/unexpected/missed ambiguity, & 2-axis summary
    res_syn = rc.evaluate_script(
        _make_spec(ex_gaps=["swap_phrase"], lang_gaps=["list_set_index"]),
        StubProvider(lambda _: (_ for _ in ()).throw(LLMProviderError("bad json"))),
        "python",
        LockfileManager(tmp_path / "isolated3.lock"),
        repo_root=tmp_path,
    )
    assert res_syn.bucket == "SYNTHESIS_ERROR"

    res_static = rc.evaluate_script(
        _make_spec(lang_gaps=["list_set_index"]),
        StubProvider(lambda _: SynthesisResult(items=[PrintStmt(expressions=[VarRefExpr(name="undecl")])])),
        "python",
        LockfileManager(tmp_path / "isolated4.lock"),
        repo_root=tmp_path,
    )
    assert res_static.bucket == "STATIC_VERIFY_GAP"

    amb_diag = AmbiguityDiagnostic(
        statement_id=1,
        source_line=1,
        source_text="amb",
        reason="amb",
        candidates=[
            AmbiguityCandidate(
                option=1,
                description="d",
                canonical_meme='print "x"',
                replacement_ir=[PrintStmt(expressions=[LiteralExpr(value="x", data_type="String")])],
            )
        ],
    )
    res_amb_ok = rc.evaluate_script(
        _make_spec(expected_status="ambiguous", expected_out=""),
        StubProvider(lambda _: SynthesisResult(items=[amb_diag])),
        "python",
        LockfileManager(tmp_path / "isolated5.lock"),
        repo_root=tmp_path,
    )
    assert res_amb_ok.bucket == "AMBIGUOUS_EXPECTED"

    res_missed = rc.evaluate_script(
        _make_spec(expected_status="ambiguous", expected_out=""),
        StubProvider(lambda _: SynthesisResult(items=[PrintStmt(expressions=[LiteralExpr(value="x", data_type="String")])])),
        "python",
        LockfileManager(tmp_path / "isolated6.lock"),
        repo_root=tmp_path,
    )
    assert res_missed.bucket == "MISSED_AMBIGUITY"

    summary = rc.summarize_results([res_mismatch, res_pass, res_syn, res_static, res_amb_ok, res_missed])
    assert summary.by_bucket["PASS"] == 1
    assert summary.by_bucket["AMBIGUOUS_EXPECTED"] == 1
    assert dict(summary.top_language_gaps)["list_set_index"] == 2
    assert dict(summary.top_examples_gaps)["swap_phrase"] == 1
    assert not (tmp_path / "build.lock").exists()
```

```python
import re


def test_full_300_uniqueness_examples_purity_and_cli_smoke(tmp_path):
    rc = _load_runner_module()
    manifest = rc.CorpusManifest.model_validate_json(
        (REPO_ROOT / "tests" / "corpus" / "manifest.json").read_text(encoding="utf-8")
    )
    assert manifest.total_scripts == 300
    assert len(manifest.scripts) == 300
    assert len({s.slug for s in manifest.scripts}) == 300
    assert len({s.title.lower() for s in manifest.scripts}) == 300

    # Anti-clone gate: stripping whitespace and numeric literals must still leave 300 unique skeletons
    skeletons = set()
    for s in manifest.scripts:
        raw = (REPO_ROOT / s.path).read_text(encoding="utf-8")
        skel = re.sub(r"\d+", "#", re.sub(r"\s+", " ", raw.strip().lower()))
        skeletons.add(skel)
    assert len(skeletons) == 300

    # `examples/` purity gate: only curated syntax/library files live in `examples/`
    example_files = sorted(p.name for p in (REPO_ROOT / "examples").glob("*.meme"))
    assert len(example_files) < 20
    assert not any(re.match(r"^\d{3}_", name) for name in example_files)

    # CLI smoke run with `--provider heuristic --limit 5 --report`, lockfile isolation, and invalid `--ids` exit code
    lock_file = REPO_ROOT / "build.lock"
    before_mtime = lock_file.stat().st_mtime_ns if lock_file.exists() else None
    report_file = tmp_path / "report.json"
    exit_ok = rc.main(
        ["--provider", "heuristic", "--limit", "5", "--report", str(report_file)]
    )
    assert exit_ok == 0
    assert report_file.is_file()
    parsed_summary = rc.CorpusRunSummary.model_validate_json(
        report_file.read_text(encoding="utf-8")
    )
    assert parsed_summary.total == 5
    after_mtime = lock_file.stat().st_mtime_ns if lock_file.exists() else None
    assert after_mtime == before_mtime

    assert rc.main(["--ids", "10-5"]) == 2
```
